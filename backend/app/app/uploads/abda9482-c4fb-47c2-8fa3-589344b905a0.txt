import numpy as np
import random
GRID_SIZE = 4
ACTIONS = ["up", "down", "left", "right"]
n_states = GRID_SIZE * GRID_SIZE
n_actions = len(ACTIONS)
rewards = np.full((GRID_SIZE, GRID_SIZE), -1)
goal = (GRID_SIZE-1, GRID_SIZE-1)
rewards[goal] = 10
def state_to_pos(state):
    return divmod(state, GRID_SIZE)
def pos_to_state(pos):
    return pos[0]*GRID_SIZE + pos[1]
Q = np.zeros((n_states, n_actions))
alpha, gamma, epsilon = 0.1, 0.9, 0.2
def step(state, action):
    row, col = state_to_pos(state)
    if action == 0 and row > 0: row -= 1
    elif action == 1 and row < GRID_SIZE-1: row += 1
    elif action == 2 and col > 0: col -= 1
    elif action == 3 and col < GRID_SIZE-1: col += 1
    new_state = pos_to_state((row, col))
    reward = rewards[row, col]
    return new_state, reward
episodes = 500

        if random.uniform(0, 1) < epsilon:
            action = random.randint(0, n_actions-1)
        else:
            action = np.argmax(Q[state])
        next_state, reward = step(state, action)
        Q[state, action] += alpha * (reward + gamma * np.max(Q[next_state]) - Q[state, action])
        state = next_state
        if state == pos_to_state(goal):
            done = True
def test_agent():
    state = 0
    path = [state]
    while state != pos_to_state(goal):
        action = np.argmax(Q[state])
        state, _ = step(state, action)
        path.append(state)
        if len(path) > 50:
            break
    return path
path = test_agent()
print("Learned Q-table:\n", Q)
print("\nOptimal path (states):", path)
print("\nOptimal path (grid positions):", [state_to_pos(s) for s in path])
